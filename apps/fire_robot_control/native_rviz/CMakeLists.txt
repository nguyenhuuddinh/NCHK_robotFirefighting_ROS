cmake_minimum_required(VERSION 3.16)
project(fire_robot_control_preview LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)

find_package(ament_cmake REQUIRED)
find_package(Qt5 REQUIRED COMPONENTS Widgets Network Test)
find_package(rclcpp REQUIRED)
find_package(rviz_common REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(tf2_msgs REQUIRED)
find_package(std_msgs REQUIRED)

add_library(fire_robot_frontend_client STATIC frontend_client.cpp frontend_client.hpp
  frontend_bootstrap.cpp frontend_bootstrap.hpp strict_json.cpp strict_json.hpp)
target_link_libraries(fire_robot_frontend_client PUBLIC Qt5::Network)
target_compile_options(fire_robot_frontend_client PRIVATE -Wall -Wextra -Wpedantic)

add_executable(fire_robot_control_preview main.cpp observer.cpp session_ui.cpp app_client.cpp app_client.hpp)
target_link_libraries(fire_robot_control_preview Qt5::Widgets fire_robot_frontend_client)
ament_target_dependencies(fire_robot_control_preview
  rclcpp rviz_common nav_msgs sensor_msgs geometry_msgs tf2_msgs std_msgs)
target_compile_options(fire_robot_control_preview PRIVATE -Wall -Wextra -Wpedantic)

enable_testing()
add_executable(fire_robot_frontend_test ../tests/qt_frontend_test.cpp)
target_link_libraries(fire_robot_frontend_test Qt5::Test fire_robot_frontend_client)
target_compile_options(fire_robot_frontend_test PRIVATE -Wall -Wextra -Wpedantic)
add_test(NAME fire_robot_frontend_test COMMAND fire_robot_frontend_test)
add_executable(fire_robot_frontend_fixture ../tests/qt_frontend_fixture.cpp)
target_link_libraries(fire_robot_frontend_fixture fire_robot_frontend_client)
target_compile_options(fire_robot_frontend_fixture PRIVATE -Wall -Wextra -Wpedantic)
add_executable(fire_robot_app_fixture ../tests/qt_app_fixture.cpp app_client.cpp app_client.hpp)
target_link_libraries(fire_robot_app_fixture fire_robot_frontend_client)
target_compile_options(fire_robot_app_fixture PRIVATE -Wall -Wextra -Wpedantic)
add_executable(fire_robot_gui_fixture main.cpp observer.cpp session_ui.cpp app_client.cpp app_client.hpp)
target_compile_definitions(fire_robot_gui_fixture PRIVATE FIRE_ROBOT_GUI_QA=1)
target_link_libraries(fire_robot_gui_fixture Qt5::Widgets fire_robot_frontend_client)
ament_target_dependencies(fire_robot_gui_fixture rclcpp rviz_common nav_msgs sensor_msgs geometry_msgs tf2_msgs std_msgs)
target_compile_options(fire_robot_gui_fixture PRIVATE -Wall -Wextra -Wpedantic)
