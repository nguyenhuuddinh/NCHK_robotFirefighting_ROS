cmake_minimum_required(VERSION 3.16)
project(fire_robot_pre_sync_ui_review LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)

  "${APP}/frontend_client.cpp" "${APP}/frontend_bootstrap.cpp")
target_include_directories(review_probe PRIVATE "${APP}")
